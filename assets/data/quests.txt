# ============================================================================
#  quests.txt - incarichi.
#
#  assegnata_da   = identificatore del tipo di entita' che la offre e la riceve
#  avanza_con     = uccisione:<entita>  |  raccolta:<oggetto>
#  bersaglio      = quante volte deve avvenire l'evento
#  consuma        = si  -> alla consegna sottrae "bersaglio" oggetti raccolti
#  sblocca        = quest che diventa disponibile alla consegna di questa
#  stato_iniziale = offerta | bloccata
#
#  L'avanzamento non e' cablato nel codice: il gioco emette gli eventi
#  "uccisione" e "raccolta", e le quest interessate reagiscono.
# ============================================================================
